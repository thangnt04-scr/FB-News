                    PHẦN 2: DOCKER WORKFLOW (5 phút)
================================================================================

# 1. Stop container hiện tại (nếu có)
docker-compose down

# 2. Build Docker image
docker-compose build

# Expected output:
# [+] Building XX.Xs (15/15) FINISHED
# Successfully built...

# 3. Start container
docker-compose up -d

# Expected output:

# ✔ Network football_football-network  Created
# ✔ Container football-app             Started

# 4. Kiểm tra container đang chạy
docker-compose ps

# Expected output:
# NAME            STATUS          PORTS
# football-app    Up (healthy)    0.0.0.0:5000->5000/tcp

# 5. Xem logs real-time
docker-compose logs -f

# Expected output:
# [INFO] Starting gunicorn 21.2.0
# [INFO] Listening at: http://0.0.0.0:5000
# [INFO] Booting worker with pid: 8
# [INFO] Booting worker with pid: 9
# [INFO] Booting worker with pid: 10
# [INFO] Booting worker with pid: 11

# Nhấn Ctrl+C để thoát logs

# 6. Test application
curl http://localhost:5000

# Hoặc mở browser:
start http://localhost:5000

# Expected: Trang web hiển thị thành công

================================================================================
                    PHẦN 3: SECURITY SCANNING (10 phút)
================================================================================

# 1. Tạo thư mục reports (nếu chưa có)
New-Item -ItemType Directory -Path reports -Force

# 2. Chạy Bandit - SAST (Static Application Security Testing)
Write-Host "Running Bandit SAST..." -ForegroundColor Yellow
bandit -r . -f json -o reports/bandit-report.json
bandit -r . -f txt -o reports/bandit-report.txt

# Expected output:
# Run started...
# Test results:
# >> Issue: [B...]
# Files skipped: X

# 3. Chạy Safety - Dependency Vulnerability Scan
Write-Host "Running Safety Dependency Scan..." -ForegroundColor Yellow
safety check --json > reports/safety-report.json
safety check > reports/safety-report.txt

# Expected output:
# Checking X packages...
# Found X known security vulnerabilities

# 4. Chạy Trivy - Container Image Scan
Write-Host "Running Trivy Container Scan..." -ForegroundColor Yellow
trivy image --format json --output reports/trivy-report.json football-app:latest
trivy image --format table --output reports/trivy-report.txt football-app:latest

# Expected output:
# Scanning football-app:latest...
# Total: X (UNKNOWN: X, LOW: X, MEDIUM: X, HIGH: X, CRITICAL: X)

# 5. Hoặc chạy tất cả bằng 1 lệnh
.\security-scan.ps1

# Expected output:
# === DEVSECOPS SECURITY SCAN ===
# [1/3] Running Bandit (SAST)...
# ✅ Bandit scan complete
# [2/3] Running Safety (Dependency)...
# ✅ Safety scan complete
# [3/3] Running Trivy (Container)...
# ✅ Trivy scan complete
# === SCAN COMPLETE ===

# 6. Xem kết quả
Get-ChildItem reports\

# Expected files:
# bandit-report.json
# bandit-report.txt
# safety-report.json
# safety-report.txt
# trivy-report.json
# trivy-report.txt

# 7. Xem chi tiết báo cáo
type reports\bandit-report.txt
type reports\safety-report.txt
type reports\trivy-report.txt

.\security-scan.ps1

Write-Host "`n=== DEMO COMPLETE ===" -ForegroundColor Green

Write-Host "Reports: reports/" -ForegroundColor Cyan

================================================================================
                    PHẦN 9: TROUBLESHOOTING NHANH
================================================================================

# Container không chạy?
docker logs football-app
docker-compose down
docker-compose build --no-cache
docker-compose up -d

# Port bị chiếm?
netstat -ano | findstr :5000
taskkill /PID <PID> /F

# Rebuild hoàn toàn?
docker-compose down -v
docker system prune -a -f
docker-compose build --no-cache
docker-compose up -d

# Xem tất cả containers?
docker ps -a

# Xem tất cả images?
docker images

# Xóa tất cả containers stopped?
docker container prune -f

# Xóa tất cả images unused?
docker image prune -a -f
